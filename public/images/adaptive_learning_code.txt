// ── EMA mastery update ──────────────────────────────────────────────────
    final newMastery = ((1 - state.alpha) * state.mastery + state.alpha * S)
        .clamp(0.0, 1.0);

    // ── Acquisition rate R ──────────────────────────────────────────────────
    // t = number of trials before
    // We track total trials; on first correct we compute R = e^(-λ*(t-1)).
    final newTotal = state.totalTrials + 1;
    final newCorrect = state.correctTrials + (isCorrect ? 1 : 0);

    double newR = state.acquisitionR;
    // Recompute R each time the correct count first crosses 1
    if (newCorrect == 1 && state.correctTrials == 0) {
      final t = newTotal; // trials until first correct
      newR = exp(-_lambda * (t - 1));
    }

    // ── Dynamic α ───────────────────────────────────────────────────────────
    final r = _beta * (newR - 0.5);
    final newAlpha = (_alpha0 + r).clamp(_alphaMin, _alphaMax);

    // ── SM-2 update ─────────────────────────────────────────────────────────
    int newN = state.sm2N;
    double newI = state.sm2I;
    double newEF = state.sm2EF;

    if (isCorrect) {
      // q = 5 for correct
      const q = 5;
      newN = state.sm2N + 1;
      if (newN == 1) {
        newI = 1;
      } else if (newN == 2) {
        newI = 6;
      } else {
        newI = (state.sm2I * state.sm2EF).roundToDouble();
      }
      newEF = (state.sm2EF + (0.1 - (5 - q) * (0.08 + (5 - q) * 0.02)))
